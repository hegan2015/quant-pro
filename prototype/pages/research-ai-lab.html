def compute(ctx):
    vol = ctx.bars("volume", 20)            # 仅 ≤ T，物理隔离
    ma20 = ctx.bars("adj_close", 20).rolling(20, min_periods=20).mean()
    pulse = vol.iloc[-1] > 1.5 * vol.mean()
    shrink = (vol.iloc[-1] < vol.iloc[-2] * 0.6) & ~vol.isna()
    hold = ctx.bars("adj_close", 1).iloc[-1] >= ma20.iloc[-1]
    return where(pulse & shrink & hold, 1.0, nan)
